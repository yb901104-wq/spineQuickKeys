    /// <summary>找本进程内可见的 Win32 模态对话框（#32770），不受外部窗口抢前台影响。</summary>
    private static IntPtr FindOwnDialog()
    {
        IntPtr found = IntPtr.Zero;
        var myPid = (uint)Environment.ProcessId;
        EnumWindows((h, _) =>
        {
            GetWindowThreadProcessId(h, out var pid);
            if (pid != myPid || !IsWindowVisible(h)) return true;
            var sb = new StringBuilder(64);
            GetClassName(h, sb, sb.Capacity);
            if (sb.ToString() != "#32770") return true;
            found = h;
            return false;
        }, IntPtr.Zero);
        return found;
    }

    private static void BackupConfig()
    {
        foreach (var f in ConfigFiles)
        {
            var p = Path.Combine(AppData, f);
            ConfigBackup[p] = File.Exists(p) ? File.ReadAllText(p) : null;
        }

    private static void RestoreConfig()
    {
        foreach (var (p, content) in ConfigBackup)
        {
            try
            {
                if (content == null) { if (File.Exists(p)) File.Delete(p); }
                else File.WriteAllText(p, content);
            }
            catch { }
        }
        Log("settings restored: True");
    }

    private static void Shot(Form w, string file, bool activate = true)
    {
        if (activate) { w.Activate(); Application.DoEvents(); }
        else Application.DoEvents();
        using var bmp = new Bitmap(Math.Max(1, w.Width), Math.Max(1, w.Height));
        using (var g = Graphics.FromImage(bmp))
            g.CopyFromScreen(w.Left, w.Top, 0, 0, bmp.Size);
        bmp.Save(Path.Combine(OutDir, file), ImageFormat.Png);
        Log("SHOT " + file);
    }

    /// <summary>
    /// MessageBox 是模态，但 WinForms 的模态循环仍会 pump WM_TIMER，
    /// 因此 Timer.Tick 可在弹窗显示期间做「运行中」断言，再发 Enter 关闭。
    /// </summary>
    private static T InvokeModal<T>(Func<T> action, string tag, Action? onModal = null)
    {
        var done = false;
        var attempts = 0;
        using var t = new System.Windows.Forms.Timer { Interval = 300 };
        t.Tick += (_, _) =>
        {
            if (done) return;
            var h = FindOwnDialog();
            if (h == IntPtr.Zero)
            {
                if (++attempts < 30) return;
                var fgsb = new StringBuilder(64);
                GetClassName(GetForegroundWindow(), fgsb, fgsb.Capacity);
                Check(false, tag + " no own dialog appeared (fg=" + fgsb + ")");
                t.Stop();
                return;
            }
            Log($"MODAL {tag}: dialog hwnd={h}");
            t.Stop();
            try { onModal?.Invoke(); }
            catch (Exception ex) { Check(false, tag + " modal probe threw: " + ex.Message); }
            // 用 WM_CLOSE 而非 SendKeys：SendKeys 在模态循环里偶发丢失导致挂死。
            SendMessage(h, WmClose, IntPtr.Zero, IntPtr.Zero);
        };
        t.Start();
        try { return action(); }
        finally { done = true; t.Stop(); }
    }

    private static void LogLayout(Form w, string tag)

        Log($"LAYOUT {tag}: win.Size={w.Size} Client={w.ClientSize} Bounds={w.Bounds}");

        Log($"LAYOUT {tag}: actionPanel boundsInPanel={ap.Bounds} size={ap.Size}");
        foreach (var n in new[] { "_btnExport", "_btnPack", "_btnExportSelected", "_btnPackSelected", "_btnSavePreset", "_btnVerifyExport" })
        {
            var b = B(w, n);
            var sc = b.PointToScreen(Point.Empty);
            Log($"LAYOUT {tag}: {n} [{b.Text}] inAP={b.Bounds} screen={sc} size={b.Size} visible={b.Visible}");
        }
        var cl = (Label)F(w, "_lblSelectionCount")!;
        Log($"LAYOUT {tag}: count screen={cl.PointToScreen(Point.Empty)} text=[{cl.Text}]");
        Log($"LAYOUT {tag}: win.Top={w.Top} win.Bottom={w.Bottom} clientBottomScreen={w.PointToScreen(new Point(0, w.ClientSize.Height)).Y}");
        Log($"LAYOUT {tag}: actionPanel AutoSize={ap.AutoSize} Wrap={ap.WrapContents} Dock={ap.Dock} Preferred={ap.PreferredSize}");
        Control? c = ep.Parent;
        while (c != null && !ReferenceEquals(c, w))
        {
            var extra = c is TableLayoutPanel t
                ? " styles=[" + string.Join(",", t.ColumnStyles.Cast<ColumnStyle>().Select(s => s.SizeType.ToString())) + "]"
                : "";
            Log($"LAYOUT {tag}: parent {c.GetType().Name} size={c.Size} AutoSize={c.AutoSize} Dock={c.Dock}{extra}");
            c = c.Parent;
        }
    }
